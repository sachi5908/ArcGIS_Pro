Action Required: Before running this code, update the yellow highlighted directory path.

  • Change the path for base_dir to the directory where your Landsat Band files are currently located.
  • Important: Make sure to add your SagarAOI.kml file into this exact same folder. If your KML has a different name, update it in the code.
  • Note: The script finds files with wildcards (LC*_B4* and LC*_B5*), so keep irrelevant TIFs out of that folder. No filenames need to be typed by hand.
  • Heads-up: This version calculates NDVI using Band 4 (Red) and Band 5 (NIR), employing the formula (NIR - Red) / (NIR + Red).
import arcpy
import os
from arcpy.sa import *

# ==========================================
# 1. SETUP & CACHE CLEARING
# ==========================================
arcpy.CheckOutExtension("Spatial")
arcpy.env.overwriteOutput = True
arcpy.ClearWorkspaceCache_management()

base_dir = r"D:\Geoinformatics\BAND Files"
arcpy.env.workspace = base_dir

print("Initializing error-proof NDVI workflow...")

# ==========================================
# 2. MOSAIC BANDS
# ==========================================
print("Mosaicking Band 4 (Red) and Band 5 (NIR)...")
b4_list = arcpy.ListRasters("LC*_B4*", "TIF")
b5_list = arcpy.ListRasters("LC*_B5*", "TIF")

arcpy.management.MosaicToNewRaster(
    input_rasters=b4_list, 
    output_location=base_dir, 
    raster_dataset_name_with_extension="Simple_B4.tif", 
    pixel_type="16_BIT_UNSIGNED", 
    number_of_bands=1
)

arcpy.management.MosaicToNewRaster(
    input_rasters=b5_list, 
    output_location=base_dir, 
    raster_dataset_name_with_extension="Simple_B5.tif", 
    pixel_type="16_BIT_UNSIGNED", 
    number_of_bands=1
)

# ==========================================
# 3. CALCULATE NDVI DIRECTLY
# ==========================================
print("Calculating NDVI directly from bands...")
b4 = Raster("Simple_B4.tif")
b5 = Raster("Simple_B5.tif")

# Standard Index Math: (NIR - Red) / (NIR + Red)
ndvi = (Float(b5) - Float(b4)) / (Float(b5) + Float(b4))

# ==========================================
# 4. CONVERT KML TO SHAPEFILE (ANY GEOMETRY)
# ==========================================
print("Converting KML to a fresh shapefile to bypass lock/corruption issues...")
kml_file = os.path.join(base_dir, "SagarAOI.kml")
kml_gdb_name = "SagarAOI_Layer"

try:
    arcpy.conversion.KMLToLayer(kml_file, base_dir, kml_gdb_name)
    gdb_path = os.path.join(base_dir, f"{kml_gdb_name}.gdb")
    
    # Dynamically walk through the GDB to find ANY feature class, regardless of type
    kml_fc = None
    for dirpath, dirnames, filenames in arcpy.da.Walk(gdb_path, datatype="FeatureClass"):
        for filename in filenames:
            kml_fc = os.path.join(dirpath, filename)
            break
        if kml_fc:
            break
            
    if not kml_fc:
        raise ValueError("No geometry found at all inside the converted KML geodatabase.")
    
    desc = arcpy.Describe(kml_fc)
    new_shp_name = "SagarAOI_Mask.shp"
    out_shp_path = os.path.join(base_dir, new_shp_name)
    
    # Intelligently handle Lines vs. Polygons
    if desc.shapeType == "Polygon":
        print("--> Found Polygon geometry. Exporting...")
        arcpy.conversion.FeatureClassToFeatureClass(kml_fc, base_dir, new_shp_name)
    elif desc.shapeType == "Polyline":
        print("--> Notice: KML contains Lines. Converting to a closed Polygon...")
        arcpy.management.FeatureToPolygon(kml_fc, out_shp_path)
    else:
        print(f"--> Found {desc.shapeType} geometry. Attempting standard export...")
        arcpy.conversion.FeatureClassToFeatureClass(kml_fc, base_dir, new_shp_name)
    
    mask_feature = out_shp_path
    print(f"--> Successfully created new mask: {mask_feature}")

except Exception as e:
    print(f"⚠️ KML to Shapefile conversion failed: {e}")
    mask_feature = os.path.join(base_dir, "sagar_polygon.shp")

# ==========================================
# 5. FAILSAFE CLIP & SAVE
# ==========================================
print("Clipping to polygon and saving to folder...")
final_output = ""

try:
    # Attempt to clip using the newly generated shapefile
    ndvi_clipped = ExtractByMask(ndvi, mask_feature)
    final_output = os.path.join(base_dir, "Simple_NDVI_Final.tif")
    ndvi_clipped.save(final_output)
    print("--> Clipping successful!")
    
except Exception as e:
    # FAILSAFE: Save the unclipped raster if clipping still fails
    print(f"⚠️ Clipping failed: {e}")
    print("--> FAILSAFE TRIGGERED: Saving the unclipped NDVI so you have the data...")
    final_output = os.path.join(base_dir, "Simple_NDVI_Unclipped.tif")
    ndvi.save(final_output)

# ==========================================
# 6. ADD TO MAP & AUTOSAVE PROJECT
# ==========================================
print("Adding final result to map and executing strict autosave...")
try:
    aprx = arcpy.mp.ArcGISProject("CURRENT")
    active_map = aprx.activeMap

    if active_map and final_output:
        active_map.addDataFromPath(final_output)

    aprx.save()
except Exception as e:
    print("--> Notice: Could not add to map/autosave (this is normal if running outside ArcGIS Pro).")

print("✅ Complete! Math processed, file saved, and ArcGIS Pro project autosaved.")

    mosaic_colormap_mode="FIRST"
)

# ==========================================
# 3. CALCULATE NDWI WITH RADIOMETRIC CALIBRATION
# ==========================================
print("Applying scaling factors and calculating NDWI_Final...")
b3 = Raster("B3_Mosaic.tif")
b5 = Raster("B5_Mosaic.tif")

# Apply Landsat scaling factor as per documentation
b3_scaled = Float(b3) * 0.0000275 - 0.2
b5_scaled = Float(b5) * 0.0000275 - 0.2

# Implement the SetNull logic from the document
invalid_mask = (b3_scaled <= 0) | (b5_scaled <= 0)
ndwi_math = (b3_scaled - b5_scaled) / (b3_scaled + b5_scaled)

ndwi_final = SetNull(invalid_mask, ndwi_math)
ndwi_final.save("NDWI_Final.tif")

# ==========================================
# 4. EXTRACT WATER AREAS (BINARY RASTER)
# ==========================================
print("Extracting potential water areas (NDWI > 0)...")
# Creates a binary raster where 1 = Water, 0 = Non-Water
water_ndwi = ndwi_final > 0
water_ndwi.save("Water_NDWI.tif")

# ==========================================
# 5. CONVERT KML TO SHAPEFILE (LOCK BYPASS)
# ==========================================
print("Preparing mask geometry...")
kml_file = os.path.join(base_dir, "SagarAOI.kml")
kml_gdb_name = "SagarAOI_Layer"
mask_feature = ""

try:
    arcpy.conversion.KMLToLayer(kml_file, base_dir, kml_gdb_name)
    gdb_path = os.path.join(base_dir, f"{kml_gdb_name}.gdb")
    
    kml_fc = None
    for dirpath, dirnames, filenames in arcpy.da.Walk(gdb_path, datatype="FeatureClass"):
        for filename in filenames:
            kml_fc = os.path.join(dirpath, filename)
            break
        if kml_fc:
            break
            
    if not kml_fc:
        raise ValueError("No geometry found.")
    
    desc = arcpy.Describe(kml_fc)
    new_shp_name = "SagarAOI_Mask.shp"
    out_shp_path = os.path.join(base_dir, new_shp_name)
    
    if desc.shapeType == "Polygon":
        arcpy.conversion.FeatureClassToFeatureClass(kml_fc, base_dir, new_shp_name)
    elif desc.shapeType == "Polyline":
        arcpy.management.FeatureToPolygon(kml_fc, out_shp_path)
    else:
        arcpy.conversion.FeatureClassToFeatureClass(kml_fc, base_dir, new_shp_name)
    
    mask_feature = out_shp_path
    print(f"--> Successfully created new mask: {mask_feature}")

except Exception as e:
    print(f"⚠️ KML to Shapefile conversion failed: {e}")
    # Final fallback if KML fails
    mask_feature = os.path.join(base_dir, "sagar_polygon.shp")

# ==========================================
# 6. FINAL CLIP (EXTRACT BY MASK)
# ==========================================
print("Clipping binary water raster to study area...")
final_output = ""

try:
    # We clip the binary Water_NDWI, not the raw NDWI_Final, as per the doc
    final_clipped = ExtractByMask(water_ndwi, mask_feature)

except Exception as e:
    print("--> Notice: Could not add to map/autosave.")

Configuration Guide

Action Required: Before running this code, you must update the yellow highlighted parts to match your local machine.

  • Change the path for arcpy.env.workspace to the folder where your unzipped Landsat TIFs are currently stored.
  • Change the output_folder path to the directory where you want the final composite images to be saved.
  • Verify if the prefix matches the exact prefix of your own Landsat files.
import arcpy
import os
import re
import sys

# Set your workspace where the unzipped Landsat TIFs are located
arcpy.env.workspace = r"D:\Geoinformatics\composite band files" 
output_folder = r"D:\Geoinformatics\test\Output"

# Ensure output directory exists
if not os.path.exists(output_folder):
    os.makedirs(output_folder)

# Define the specific scene prefix
prefix = "LC09_L2SP_145043_20251225_20251227_02_T1"

# 1. AUTOMATICALLY DETECT AVAILABLE BANDS
available_bands = {}
for file in os.listdir(arcpy.env.workspace):
    if file.startswith(prefix) and file.upper().endswith(".TIF"):
        match = re.search(r'_SR_B(\d+)', file, re.IGNORECASE)
        if match:
            band_num = int(match.group(1))
            if 1 <= band_num <= 7:
                available_bands[band_num] = file

if not available_bands:
    print("🚨 No bands found for this prefix. Check your folder path and prefix.")
    sys.exit()

# Sort to ensure consistent order for the master composite
sorted_band_nums = sorted(available_bands.keys())
print(f"Detected bands: {sorted_band_nums}")

# 2. DEFINE DESIRED COMBINATIONS (Strict RGB Order)
combinations = {
    "TCC_432": [4, 3, 2],
    "Veg_FCC_543": [5, 4, 3],
    "VegMoist_654": [6, 5, 4],
    "Geology_764": [7, 6, 4],
    "VegSoil_753": [7, 5, 3],
    "MoistVeg_652": [6, 5, 2],
    "GeoSoil_752": [7, 5, 2],
    "VegMoistContrast_562": [5, 6, 2]
}

print("\n--- Phase 1: Generating specific 3-band TIF files ---")

# 3. GENERATE PHYSICAL 3-BAND FILES AND RENAME BANDS
for name, bands in combinations.items():
    if all(b in available_bands for b in bands):
        
        in_rasters = [os.path.join(arcpy.env.workspace, available_bands[b]) for b in bands]
        in_rasters_str = ";".join(in_rasters)
        out_raster = os.path.join(output_folder, f"{prefix}_{name}.tif")
        
        if not os.path.exists(out_raster):
            print(f"Creating {name}...")
            
            # Run CompositeBands
            arcpy.management.CompositeBands(in_rasters_str, out_raster)
            
            # Rename internal bands to avoid UI confusion and collision bugs
            rast_obj = arcpy.Raster(out_raster)
            for i, b_num in enumerate(bands):
                default_name = f"Band_{i+1}"
                target_name = f"L9_Band_{b_num}" 
                try:
                    rast_obj.renameBand(default_name, target_name)
                except Exception as e:
                    print(f"   ⚠️ Could not rename {default_name} to {target_name}: {e}")
            
            # Calculate statistics for perfect immediate contrast
            arcpy.management.CalculateStatistics(out_raster)
            print(f"✅ Saved: {out_raster}")
            
        else:
            print(f"⚠️ File already exists, skipping: {out_raster}")
    else:
        print(f"❌ Skipped {name}: Missing one or more required bands {bands}")


print("\n--- Phase 2: Generating the Master Dynamic Composite ---")

# 4. GENERATE THE MASTER ALL-BAND COMPOSITE
master_out_raster = os.path.join(output_folder, f"{prefix}_DynamicComposite.tif")

if not os.path.exists(master_out_raster):
    print("Creating Master Dynamic Composite (All available bands)...")
    
    # Construct input string using all detected bands in numerical order
    master_in_rasters = [os.path.join(arcpy.env.workspace, available_bands[b]) for b in sorted_band_nums]
    master_in_rasters_str = ";".join(master_in_rasters)
    
    # Run CompositeBands for the master file
    arcpy.management.CompositeBands(master_in_rasters_str, master_out_raster)
    
    # Rename internal bands for the master file
    master_rast_obj = arcpy.Raster(master_out_raster)
    for i, b_num in enumerate(sorted_band_nums):
        default_name = f"Band_{i+1}"
        target_name = f"L9_Band_{b_num}" 
        try:
            master_rast_obj.renameBand(default_name, target_name)
        except Exception as e:
            print(f"   ⚠️ Could not rename {default_name} to {target_name}: {e}")
            
    # Calculate statistics
    arcpy.management.CalculateStatistics(master_out_raster)
    print(f"✅ Saved Master Composite: {master_out_raster}")
else:
    print(f"⚠️ Master file already exists, skipping: {master_out_raster}")

print("\nWorkflow completely finished! All specific composites and your master file are ready.")

Configuration Guide

Action Required: Before running this code, update the yellow highlighted paths.

  • Change input_dem to your DEM file (the current one is an SRTM 1 arc-second tile, processed in UTM 44N).
  • Change output_root to the folder where results should be saved. Each run creates its own timestamped sub-folder containing outputs, run.log and Watershed_Results.lyrx.
  • Check that propy points to your ArcGIS Pro propy.bat launcher (the default install path is already set).
  • Before running: open the 2D map that should receive the results and make it the active map tab. Save your work first. The project is not saved automatically.
  • Tip: threshold of 50,000 cells gives only major streams. Lower it (for example 5,000) for more tributaries and sub-basins. show_rasters controls whether the watershed rasters are included.
  • Paste the whole script into the ArcGIS Pro Python window. Processing runs in a separate Python process and takes about 1-2 minutes; five layers are then added to the active map. If map insertion fails, add Watershed_Results.lyrx manually with Add Data.

WATERSHED_WORKER_SOURCE = r'''
import datetime as dt
import json
from pathlib import Path
import sys
import arcpy as ap
from arcpy.sa import (Con, Fill, FlowAccumulation, FlowDirection,
                      StreamLink, StreamOrder, StreamToFeature, Watershed)

def log(message):
    print(f'[{dt.datetime.now().isoformat(timespec="seconds")}] {message}', flush=True)

def raster(value, path):
    value.save(str(path))
    return str(path)

def process(config):
    run = Path(config['run_dir'])
    out = run / 'outputs'
    scratch = run / 'scratch'
    out.mkdir()
    scratch.mkdir()
    if not ap.Exists(config['input_dem']):
        raise FileNotFoundError(config['input_dem'])
    if ap.Describe(config['input_dem']).spatialReference.name == 'Unknown':
        raise ValueError('The DEM must have a known coordinate system.')
    if ap.CheckExtension('Spatial') != 'Available':
        raise RuntimeError('Spatial Analyst is unavailable.')
    if ap.ProductInfo() not in ('ArcInfo', 'ArcEditor'):
        raise RuntimeError('SmoothLine requires a Standard or Advanced license.')
    if ap.CheckOutExtension('Spatial') != 'CheckedOut':
        raise RuntimeError('Could not check out Spatial Analyst.')
    # Deliberately no CheckInExtension: that call crashes on this installation.
    # Esri returns the license when this SHORT-LIVED standalone worker ends.
    # Do not paste/run this worker directly in the long-lived Pro Python window.
    with ap.EnvManager(workspace=str(out), scratchWorkspace=str(scratch),
                       overwriteOutput=False, addOutputsToMap=False,
                       parallelProcessingFactor='50%', compression='LZW'):
        log('1/13 Projecting DEM to UTM 44N')
        projected = str(out / 'DEM_UTM44N.tif')
        ap.management.ProjectRaster(config['input_dem'], projected,
                                    ap.SpatialReference(32644), 'NEAREST')
        with ap.EnvManager(snapRaster=projected, cellSize=projected,
                           extent=projected, outputCoordinateSystem=ap.SpatialReference(32644)):
            log('2/13 Filling sinks')
            filled = raster(Fill(projected), scratch / 'Filled_DEM.tif')
            log('3/13 Flow Direction (D8)')
            direction = raster(FlowDirection(filled, force_flow='NORMAL',
                                             flow_direction_type='D8'),
                               scratch / 'Flow_Direction.tif')
            log('4/13 Flow Accumulation')
            accumulation = raster(FlowAccumulation(direction, data_type='FLOAT',
                                                   flow_direction_type='D8'),
                                  scratch / 'Flow_Accumulation.tif')
            maximum = float(ap.management.GetRasterProperties(accumulation, 'MAXIMUM')[0])
            if maximum <= config['threshold']:
                raise ValueError('Threshold is too high: no stream cells would be extracted.')
            log('5/13 Extracting stream network')
            streams = raster(Con(ap.Raster(accumulation) > config['threshold'], 1),
                             scratch / 'Streams.tif')
            log('6/13 Strahler stream order')
            order = raster(StreamOrder(streams, direction, 'STRAHLER'),
                           scratch / 'Stream_Order.tif')
            log('7/13 Converting streams to features')
            raw = str(out / 'Stream_Network_Raw.shp')
            StreamToFeature(order, direction, raw, 'NO_SIMPLIFY')
            log('8/13 Smoothing stream lines')
            smooth = str(out / 'Stream_Network_Smoothed.shp')
            ap.cartography.SmoothLine(raw, smooth, 'PAEK', config['smoothing'])
            log('9/13 Stream links')
            links = raster(StreamLink(streams, direction), scratch / 'Stream_Links.tif')
            log('10/13 Tributary sub-basins')
            sub_raster = raster(Watershed(direction, links), out / 'Sub_Basins_Ras.tif')
            sub_vector = str(out / 'Sub_Basins.shp')
            ap.conversion.RasterToPolygon(sub_raster, sub_vector, 'NO_SIMPLIFY', 'Value')
            log(f'11/13 Maximum-flow outlet: {maximum}')
            outlet = raster(Con(ap.Raster(accumulation) == maximum, 1), scratch / 'Pour_Point.tif')
            log('12/13 Main watershed')
            main_raster = raster(Watershed(direction, outlet), out / 'Main_Watershed.tif')
            main_vector = str(out / 'Main_Basin.shp')
            ap.conversion.RasterToPolygon(main_raster, main_vector, 'SIMPLIFY', 'Value')
            log('13/13 Clipping sub-basins and streams')
            ap.analysis.Clip(sub_vector, main_vector, str(out / 'Clipped_Sub_Basins.shp'))
            ap.analysis.Clip(smooth, main_vector, str(out / 'Clipped_Streams.shp'))
    counts = {name: int(ap.management.GetCount(str(out / name))[0]) for name in
              ('Main_Basin.shp', 'Clipped_Sub_Basins.shp', 'Clipped_Streams.shp')}
    if not all(counts.values()):
        raise RuntimeError(f'Unexpected empty output: {counts}')
    (run / 'processing.json').write_text(json.dumps(counts, indent=2), encoding='utf-8')
    log('Processing saved; worker exiting normally to release its license.')

def prepare_layers(config):
    run = Path(config['run_dir'])
    folder = run / 'layers'
    folder.mkdir()
    specifications = [
        ('Main_Watershed.tif', 'Main Watershed (Raster)', None, 50),
        ('Sub_Basins_Ras.tif', 'Sub-Basin Watersheds (Raster)', None, 40),
        ('Main_Basin.shp', 'Main Watershed Boundary', None, 0),
        ('Clipped_Sub_Basins.shp', 'Tributary Sub-Basins', 'gridcode', 40),
        ('Clipped_Streams.shp', 'Stream Network', 'grid_code', 0),
    ]
    bundle = None
    with ap.EnvManager(addOutputsToMap=False, overwriteOutput=False):
        for index, (name, title, field, transparency) in enumerate(specifications):
            log(f'Preparing saved layer: {title}')
            source = str(run / 'outputs' / name)
            temporary = f'watershed_layer_{index}'
            is_raster = name.endswith('.tif')
            make = ap.management.MakeRasterLayer if is_raster else ap.management.MakeFeatureLayer
            result = make(source, temporary)
            path = folder / (Path(name).stem + '.lyrx')
            ap.management.SaveToLayerFile(result.getOutput(0), str(path), 'ABSOLUTE')
            layer_file = ap.mp.LayerFile(str(path))
            layer = layer_file.listLayers()[0]
            layer.name = title
            layer.transparency = transparency
            if is_raster:
                layer.visible = config['show_rasters']
            else:
                sym = layer.symbology
                if field:
                    if field not in {f.name for f in ap.ListFields(source)}:
                        raise ValueError(f'Missing field {field} in {source}')
                    sym.updateRenderer('UniqueValueRenderer')
                    sym.renderer.fields = [field]
                else:
                    sym.updateRenderer('SimpleRenderer')
                    sym.renderer.symbol.color = {'RGB': [0, 0, 0, 0]}
                    sym.renderer.symbol.outlineColor = {'RGB': [0, 0, 0, 100]}
                    sym.renderer.symbol.outlineWidth = 1.5
                layer.symbology = sym
            layer_file.save()
            if bundle is None:
                bundle = layer_file
            else:
                bundle.addLayer(layer_file, 'TOP')
            ap.management.Delete(temporary)
        bundle_path = run / 'Watershed_Results.lyrx'
        bundle.saveACopy(str(bundle_path))
        check = ap.mp.LayerFile(str(bundle_path))
        if len(check.listLayers()) != 5 or check.listBrokenDataSources():
            raise RuntimeError('Saved layer collection is incomplete or has broken sources.')
    (run / 'layers.json').write_text(json.dumps({'layer_file': str(bundle_path)}, indent=2),
                                    encoding='utf-8')
    log('Five styled layers prepared without opening or creating a project.')

if __name__ == '__main__':
    config = json.loads(Path(sys.argv[2]).read_text(encoding='utf-8'))
    log(f'ArcGIS Python: {sys.executable}')
    try:
        if sys.argv[1] == 'process':
            process(config)
        elif sys.argv[1] == 'layers':
            prepare_layers(config)
        else:
            raise ValueError('Unknown worker phase.')

        traceback.print_exc()

def watershed_in_current_project():
    import datetime as dt
    import json
    import math
    import os
    from pathlib import Path
    import subprocess
    import uuid
    import arcpy as ap

    # EDIT THESE SETTINGS IF NEEDED. UTM 44N is retained for this supplied DEM.
    input_dem = Path(r'C:\Users\sachi\OneDrive\Desktop\check\Test_Data\input\n24_e079_1arc_v3.tif')
    output_root = Path(r'C:\Users\sachi\OneDrive\Desktop\check\Test_Data\current_project_runs')
    threshold = 50000
    smoothing = '90 Meters'
    show_rasters = True
    propy = Path(r'C:\Program Files\ArcGIS\Pro\bin\Python\Scripts\propy.bat')

    # Capture the actual current project/map; no hardcoded APRX or project copy.
    project = ap.mp.ArcGISProject('CURRENT')
    target = project.activeMap
    if target is None:
        raise RuntimeError('Open the destination map tab, then paste/run this code again.')
    if target.mapType != 'MAP':
        raise RuntimeError('Open a 2D map, not a 3D scene, before running this workflow.')
    if not input_dem.is_file():
        raise FileNotFoundError(f'Input DEM not found: {input_dem}')
    if not propy.is_file():
        raise FileNotFoundError(f'ArcGIS Python launcher not found: {propy}')
    if not math.isfinite(float(threshold)) or threshold <= 0:
        raise ValueError('Threshold must be positive and finite.')

    run = output_root / (dt.datetime.now().strftime('%Y%m%d_%H%M%S')
                         + '_' + uuid.uuid4().hex[:8])
    run.mkdir(parents=True)
    worker = run / 'worker.py'
    worker.write_text(WATERSHED_WORKER_SOURCE, encoding='utf-8')
    config_path = run / 'job.json'
    config_path.write_text(json.dumps({
        'run_dir': str(run), 'input_dem': str(input_dem), 'threshold': threshold,
        'smoothing': smoothing, 'show_rasters': show_rasters,
        'current_project': project.filePath, 'current_map': target.name,
    }, indent=2), encoding='utf-8')
    status = {'state': 'running', 'project': project.filePath,
              'map': target.name, 'run_dir': str(run), 'phases': {}}

    def write_status():
        temporary = run / 'status.tmp'
        temporary.write_text(json.dumps(status, indent=2), encoding='utf-8')
        temporary.replace(run / 'status.json')

    environment = os.environ.copy()
    for key in ('PYTHONHOME', 'PYTHONPATH'):
        environment.pop(key, None)
    environment['PYTHONUTF8'] = '1'
    environment['PYTHONUNBUFFERED'] = '1'
    command_shell = os.environ.get('COMSPEC', r'C:\Windows\System32\cmd.exe')
    write_status()
    print(f'Running watershed for CURRENT map: {target.name}', flush=True)
    print(f'Outputs and log: {run}', flush=True)
    print('Please wait. Processing is isolated; the Python window remains busy.', flush=True)
    try:
        with (run / 'run.log').open('w', encoding='utf-8', buffering=1) as log_file:
            for phase in ('process', 'layers'):
                status['phase'] = phase
                write_status()
                # Raw cmd command line avoids incorrect list2cmdline escaping
                # of the nested quotes around the installed propy.bat path.
                command = f'"{command_shell}" /d /s /c ""{propy}" -u "{worker}" {phase} "{config_path}""'
                with subprocess.Popen(command, cwd=str(run), env=environment,
                                      stdin=subprocess.DEVNULL, stdout=subprocess.PIPE,
                                      stderr=subprocess.STDOUT, text=True,
                                      encoding='utf-8', errors='replace',
                                      creationflags=subprocess.CREATE_NO_WINDOW) as child:
                    for line in child.stdout:
                        log_file.write(line)
                        print(line.rstrip(), flush=True)
                    code = child.wait()
                status['phases'][phase] = {'exit_code': code,
                                           'exit_hex': f'0x{code & 0xffffffff:08X}'}
                if code != 0:
                    raise RuntimeError(f'{phase} worker failed (0x{code & 0xffffffff:08X}). '
                                       f'See {run / "run.log"}')
                manifest = run / ('processing.json' if phase == 'process' else 'layers.json')
                if not manifest.is_file():
                    raise RuntimeError(f'{phase} worker did not produce its completion manifest.')

            # Only now touch the live map, after both child processes exited.
            # Styles were saved OFFLINE; no live renderer/gallery updates.
            status['phase'] = 'adding_to_current_map'
            write_status()
            message = f'BEFORE adding prepared layers to CURRENT map: {target.name}'
            log_file.write(message + '\n')
            print(message, flush=True)
            layer_file = ap.mp.LayerFile(str(run / 'Watershed_Results.lyrx'))
            if len(layer_file.listLayers()) != 5 or layer_file.listBrokenDataSources():
                raise RuntimeError('Prepared layer file is incomplete or has broken sources.')
            with ap.EnvManager(addOutputsToMap=False):
                added = target.addLayer(layer_file, 'TOP')
            status['added_layers'] = [layer.name for layer in added]
            if len(added) != 5 or any(layer.isBroken for layer in added):
                raise RuntimeError('Not all five result layers were added correctly.')
            log_file.write('AFTER adding five prepared layers to CURRENT map\n')
        status['state'] = 'succeeded'
        status['project_saved'] = False
        write_status()
        print(f'DONE: five layers added to your existing map "{target.name}".', flush=True)
        print('Press Ctrl+S to save this same project when ready.', flush=True)
        print(f'Outputs: {run / "outputs"}', flush=True)
        return str(run)
    except Exception as error:
        status['state'] = 'failed'
        status['error'] = str(error)
        write_status()
        print(f'FAILED: {error}', flush=True)
        print(f'Saved outputs/logs remain in: {run}', flush=True)
        print('If layers were prepared, use Add Data with Watershed_Results.lyrx.', flush=True)
        raise


watershed_run_folder = watershed_in_current_project()
